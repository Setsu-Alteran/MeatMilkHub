Основна схема (виокремлені таблиці)
Сутності:
categories — категорії продуктів

manufacturers — виробники

products — базові продукти

product_variants — варіанти (якщо потрібно: різні упаковки/веса)

prices — історія/відстеження цін (опціонально)

inventory — поточні залишки по варіантах/локаціях

purchase_orders — закупівлі від виробників

purchase_order_items — рядки закупівель

sales_orders — продажі клієнтам

sales_order_items — рядки продажів


-- Категорії
CREATE TABLE categories (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL UNIQUE
);

-- Виробники
CREATE TABLE manufacturers (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL UNIQUE
);

-- Продукти
CREATE TABLE products (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL,
  category_id INT NOT NULL REFERENCES categories(id) ON DELETE RESTRICT,
  manufacturer_id INT NOT NULL REFERENCES manufacturers(id) ON DELETE RESTRICT,
  description TEXT,
  sku TEXT UNIQUE, -- артикул, якщо потрібен
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- Варіанти продукту (упаковка, вага тощо)
CREATE TABLE product_variants (
  id SERIAL PRIMARY KEY,
  product_id INT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
  variant_name TEXT NOT NULL, -- "1 кг", "250 г", "пачка"
  barcode TEXT UNIQUE,
  unit VARCHAR(50), -- одиниця виміру
  multiplier NUMERIC, -- якщо потрібно (наприклад, 1 = базова одиниця)
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- Інвентар (за варіантами, можливо по локаціях)
CREATE TABLE inventory (
  id SERIAL PRIMARY KEY,
  product_variant_id INT NOT NULL REFERENCES product_variants(id) ON DELETE CASCADE,
  quantity NUMERIC NOT NULL DEFAULT 0,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  UNIQUE(product_variant_id) -- якщо без локацій
);

-- Закупівельні замовлення
CREATE TABLE purchase_orders (
  id SERIAL PRIMARY KEY,
  manufacturer_id INT NOT NULL REFERENCES manufacturers(id),
  order_date DATE NOT NULL DEFAULT CURRENT_DATE,
  expected_arrival DATE,
  status VARCHAR(30) NOT NULL DEFAULT 'pending', -- pending, received, cancelled
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE TABLE purchase_order_items (
  id SERIAL PRIMARY KEY,
  purchase_order_id INT NOT NULL REFERENCES purchase_orders(id) ON DELETE CASCADE,
  product_variant_id INT NOT NULL REFERENCES product_variants(id),
  quantity NUMERIC NOT NULL,
  unit_price NUMERIC NOT NULL,
  total_price NUMERIC GENERATED ALWAYS AS (quantity * unit_price) STORED
);

-- Продажі
CREATE TABLE sales_orders (
  id SERIAL PRIMARY KEY,
  customer_name TEXT,
  order_date DATE NOT NULL DEFAULT CURRENT_DATE,
  status VARCHAR(30) NOT NULL DEFAULT 'draft', -- draft, confirmed, shipped, cancelled
  total_amount NUMERIC DEFAULT 0,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE TABLE sales_order_items (
  id SERIAL PRIMARY KEY,
  sales_order_id INT NOT NULL REFERENCES sales_orders(id) ON DELETE CASCADE,
  product_variant_id INT NOT NULL REFERENCES product_variants(id),
  quantity NUMERIC NOT NULL,
  unit_price NUMERIC NOT NULL,
  total_price NUMERIC GENERATED ALWAYS AS (quantity * unit_price) STORED
);



interface Category {
  id: number;
  name: string;
}

interface Manufacturer {
  id: number;
  name: string;
}

interface Product {
  id: number;
  name: string;
  categoryId: number;
  manufacturerId: number;
  description?: string;
  sku?: string;
  createdAt: string; // ISO timestamp
  updatedAt: string;
}

interface ProductVariant {
  id: number;
  productId: number;
  variantName: string;
  barcode?: string;
  unit?: string;
  multiplier?: number;
  createdAt: string;
}

interface Inventory {
  id: number;
  productVariantId: number;
  quantity: number;
  updatedAt: string;
}

interface PurchaseOrderItem {
  id: number;
  purchaseOrderId: number;
  productVariantId: number;
  quantity: number;
  unitPrice: number;
  totalPrice: number;
}

interface PurchaseOrder {
  id: number;
  manufacturerId: number;
  orderDate: string;
  expectedArrival?: string;
  status: 'pending' | 'received' | 'cancelled';
  createdAt: string;
  items?: PurchaseOrderItem[];
}

interface SalesOrderItem {
  id: number;
  salesOrderId: number;
  productVariantId: number;
  quantity: number;
  unitPrice: number;
  totalPrice: number;
}

interface SalesOrder {
  id: number;
  customerName?: string;
  orderDate: string;
  status: 'draft' | 'confirmed' | 'shipped' | 'cancelled';
  totalAmount: number;
  createdAt: string;
  items?: SalesOrderItem[];
}
